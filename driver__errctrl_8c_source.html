52void ERRCTRL_ModifyControlA(uint8_t groupMask, uint8_t groupConfig)
53{
54 const uint8_t currentValue = ERRCTRL.CTRLA;
55 const uint8_t clearedConfig = currentValue & (uint8_t)(~groupMask);
