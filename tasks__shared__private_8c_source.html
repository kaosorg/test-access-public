51void ReportClearBuserrChannelError(void)
52{
53 if (MW_GetErrorChannel(ERRCH_BUSERR) != NO_ERROR)
54 {
55 errFlag_t flag = MW_GetClearBusError();
56 EH_HandleError(flag, ERRID_BUSERR_BUSERR);
57
58 // Check and report if bus error was due to bus parity error (ECC errors are reported and
59 // cleared separately)
60 if (flag == ERROR)
61 {
62 flag = MW_GetClearRamParityError();

65 flag = MW_GetClearNvmParityError();
